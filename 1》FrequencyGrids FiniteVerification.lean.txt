/-- Borne numérique : `log 5000 ≤ 8.55` via `5000^3 ≤ 2^37`. -/
lemma log_5000_le : Real.log 5000 ≤ 8.55 := by
  have hpow : (5000 : ℝ) ^ 3 ≤ (2 : ℝ) ^ 37 := by norm_num
  have hlog_pow : 3 * Real.log 5000 ≤ 37 * Real.log 2 := by
    have h1 : Real.log ((5000 : ℝ) ^ 3) ≤ Real.log ((2 : ℝ) ^ 37) :=
      Real.log_le_log (by positivity) hpow

  have hlog2 := log_two_bounds.2
  linarith

/-- Borne supérieure `aOf t ≤ Delta / 4` pour tout `t : ℕ`. -/
lemma aOf_le_delta_div_four (t : ℕ) : aOf t ≤ Delta / 4 := by
  unfold aOf Delta
  have hlog_pos : 0 < Real.log 2 := log_two_pos
  have hpi_pos : 0 < Real.pi := Real.pi_pos
  rw [div_le_div_iff₀ hlog_pos (by positivity)]
  calc Real.arcsin |s2 t| * (4 * Real.log 2)
      ≤ (Real.pi / 2) * (4 * Real.log 2) := by
        gcongr
        exact Real.arcsin_le_pi_div_two |s2 t|
    _ = (2 * Real.pi / Real.log 2) * Real.log 2 / 4 * (4 * Real.log 2) := by
        have hne : Real.log 2 ≠ 0 := hlog_pos.ne'
        field_simp [hne]
        ring
    _ = (2 * Real.pi / Real.log 2) / 4 * (Real.log 2 * (4 * Real.log 2)) := by
        have hne : Real.log 2 ≠ 0 := hlog_pos.ne'
        field_simp [hne]
        ring

/-- Minoration `Delta / 6 ≤ aOf 1`, déduite de `arcsin_abs_s2_ge` pour `t = 1`. -/
lemma delta_div_six_le_aOf_one : Delta / 6 ≤ aOf 1 := by
  unfold aOf Delta
  have h_arc : Real.pi / 3 ≤ Real.arcsin |s2 1| :=
    arcsin_abs_s2_ge 1 (by norm_num) (by norm_num)
  have hlog_pos : 0 < Real.log 2 := log_two_pos
  have hpi_pos : 0 < Real.pi := Real.pi_pos
  rw [div_le_div_iff₀ (by positivity) hlog_pos]
  calc (2 * Real.pi / Real.log 2) / 6 * Real.log 2
      = Real.pi / 3 := by
        have hne : Real.log 2 ≠ 0 := hlog_pos.ne'
        field_simp [hne]
        ring
    _ ≤ Real.arcsin |s2 1| := h_arc

def EdgeOne (T : ℝ) : Prop :=
  ∀ (k : ℤ) (σ : ℝ), σ ∈ sigmaSet → σ * (k : ℝ) < 0 → |b 1 k σ| ≤ T → |b 2 k σ| ≤ T

def EdgeTen (T : ℝ) : Prop :=
  ∀ (k : ℤ) (σ : ℝ), σ ∈ sigmaSet → 0 < σ * (k : ℝ) → |b 10 k σ| ≤ T → |b 9 k σ| ≤ T

/-- Fermeture du bord `(t = 10, σ * k > 0)` lorsque `n * Δ ≤ T < (n + 1) * Δ - Δ / 4`. -/
theorem edgeTen_of_gap (T : ℝ) (n : ℕ)
    (h1 : (n : ℝ) * Delta ≤ T) (h2 : T < ((n : ℝ) + 1) * Delta - Delta / 4) :
    EdgeTen T := by
  intro k σ hσ hpos hbT
  have hk_ne : k ≠ 0 := by
    rintro rfl
    simp at hpos
  have h_signed10 := (abs_b_signed 10 k hk_ne σ hσ).1 hpos
  have h_signed9 := (abs_b_signed 9 k hk_ne σ hσ).1 hpos
  have h_a10 : aOf 10 ≤ Delta / 4 := aOf_le_delta_div_four 10
  have h_a9_nonneg : 0 ≤ aOf 9 := by
    unfold aOf
    exact div_nonneg (Real.arcsin_nonneg.mpr (abs_nonneg _)) log_two_pos.le
  have hm_eq : ((k.natAbs : ℕ) : ℝ) = |(k : ℝ)| := by
    simp [Int.natAbs_cast]
  have hm_le_n : k.natAbs ≤ n := by
    by_contra h_gt
    push_neg at h_gt
    have hm_ge : ((n : ℝ) + 1) ≤ ((k.natAbs : ℕ) : ℝ) := by
      exact_mod_cast h_gt
    have h_delta_pos : 0 < Delta := Delta_pos
    have h_ge : ((n : ℝ) + 1) * Delta - Delta / 4 ≤ |b 10 k σ| := by
      rw [h_signed10, ← hm_eq]
      nlinarith
    linarith
  have hm_le_n_real : ((k.natAbs : ℕ) : ℝ) ≤ (n : ℝ) := by exact_mod_cast hm_le_n
  rw [h_signed9, ← hm_eq]
  nlinarith

/-- Déduction conditionnelle : `N_off = o(N_R)` et `N_van = o(N_R)` ⟹ `η_R → 1`. -/
theorem etaR_tendsto_one_of_littleO
    (X : ℝ → ℕ) (δ : ℝ → ℝ)
    (h_off : (fun T => (Noff T (X T) (δ T) : ℝ)) =o[atTop] (fun T => (NR T : ℝ)))
    (h_van : (fun T => (Nvan T (X T) (δ T) : ℝ)) =o[atTop] (fun T => (NR T : ℝ))) :
    Tendsto (fun T => etaR T (X T) (δ T)) atTop (𝓝 1) := by
  have h_div : Tendsto
      (fun T => ((Noff T (X T) (δ T) : ℝ) + (Nvan T (X T) (δ T) : ℝ)) / (NR T : ℝ))
      atTop (𝓝 0) := (h_off.add h_van).tendsto_div_nhds_zero
  have h_lim : Tendsto
      (fun T => 1 - ((Noff T (X T) (δ T) : ℝ) + (Nvan T (X T) (δ T) : ℝ)) / (NR T : ℝ))
      atTop (𝓝 1) := by
    simpa using tendsto_const_nhds.sub h_div
  exact h_lim.congr fun T => (etaR_eq T (X T) (δ T)).symm

/-- Réciproque partielle : `η_R → 1` ⟹ `N_van = o(N_R)`. Aucune hypothèse sur `δ`. -/
theorem nvan_littleO_of_etaR_tendsto_one
    (X : ℝ → ℕ) (δ : ℝ → ℝ)
    (h : Tendsto (fun T => etaR T (X T) (δ T)) atTop (𝓝 1)) :
    (fun T => (Nvan T (X T) (δ T) : ℝ)) =o[atTop] (fun T => (NR T : ℝ)) := by
  have hNR : ∀ᶠ T in atTop, (0 : ℝ) < (NR T : ℝ) := by
    filter_upwards [eventually_NR_pos] with T hT
    exact Nat.cast_pos.mpr hT
  have hsum : Tendsto
      (fun T => ((Noff T (X T) (δ T) : ℝ) + (Nvan T (X T) (δ T) : ℝ)) / (NR T : ℝ))
      atTop (𝓝 0) := by
    have h1 := (tendsto_const_nhds (x := (1 : ℝ))).sub h
    rw [sub_self] at h1
    refine h1.congr fun T => ?_
    rw [etaR_eq]
    ring
  have hvan : Tendsto (fun T => (Nvan T (X T) (δ T) : ℝ) / (NR T : ℝ)) atTop (𝓝 0) := by
    refine squeeze_zero'
      (hNR.mono fun T hT => div_nonneg (Nat.cast_nonneg _) hT.le)
      (hNR.mono fun T hT => ?_)
      hsum
    have h_le : (Nvan T (X T) (δ T) : ℝ) ≤
        (Noff T (X T) (δ T) : ℝ) + (Nvan T (X T) (δ T) : ℝ) := by
      linarith [(Nat.cast_nonneg (Noff T (X T) (δ T)) : (0 : ℝ) ≤ _)]
    exact div_le_div_of_nonneg_right h_le hT.le
  exact (isLittleO_iff_tendsto' (hNR.mono fun T hT h0 => absurd h0 hT.ne')).mpr hvan

/-- Minoration `44.12 * T - 600 ≤ (NR T : ℝ)` pour `T ≥ 0`, déduite de `Ntotal_approx`
et `NR_eq_Ntotal`. -/
lemma nr_ge_linear_sub (T : ℝ) (hT : 0 ≤ T) :
    (44.12 : ℝ) * T - 600 ≤ (NR T : ℝ) := by
  have h_approx := (abs_le.mp (Ntotal_approx T hT)).1
  have h_nr_eq : (NR T : ℝ) = (Ntotal T : ℝ) := by
    exact_mod_cast NR_eq_Ntotal T
  linarith

/-- Négation de `Eta_Tends_ToOne` : le témoin `X(T) = ⌊√T⌋` et `δ(T) = 1 / T`
vérifie `X → +∞` et `δ → 0`, mais `η_R` ne tend pas vers `1` car cela imposerait
`N_van = o(N_R)`, contredit par `X(T) ≤ (1/2) * N_R(T)` à terme. -/
theorem not_Eta_Tends_ToOne : ¬ Eta_Tends_ToOne := by
  intro h_conj
  let X : ℝ → ℕ := fun T => ⌊Real.sqrt T⌋₊
  let δ : ℝ → ℝ := fun T => 1 / T
  have hX : Tendsto (fun T => (X T : ℝ)) atTop atTop := by
    have hsqrt : Tendsto Real.sqrt atTop atTop := Real.tendsto_sqrt_atTop
    refine tendsto_atTop_mono (fun T => ?_) (hsqrt.atTop_add tendsto_const_nhds)
    have hfl : (Real.sqrt T : ℝ) - 1 < (X T : ℝ) :=
      Nat.sub_one_lt_floor (Real.sqrt T)
    linarith
  have hδ : Tendsto δ atTop (𝓝 0) :=
    tendsto_const_nhds.div_atTop tendsto_id
  have hδ_nonneg : ∀ᶠ T in atTop, 0 ≤ δ T := by
    filter_upwards [eventually_ge_atTop (0 : ℝ)] with T hT
    exact div_nonneg zero_le_one hT
  have hη : Tendsto (fun T => etaR T (X T) (δ T)) atTop (𝓝 1) :=
    h_conj X δ hX hδ
  have hvan : (fun T => (Nvan T (X T) (δ T) : ℝ)) =o[atTop] (fun T => (NR T : ℝ)) :=
    nvan_littleO_of_etaR_tendsto_one X δ hη
  have hle : ∀ᶠ T in atTop, (X T : ℝ) ≤ (1 / 2 : ℝ) * (NR T : ℝ) := by
    filter_upwards [eventually_ge_atTop (100 : ℝ)] with T hT
    have hT0 : 0 ≤ T := by linarith
    have h_nr_lo : (44.12 : ℝ) * T - 600 ≤ (NR T : ℝ) := nr_ge_linear_sub T hT0
    have h_floor : (X T : ℝ) ≤ Real.sqrt T := Nat.floor_le (Real.sqrt_nonneg T)
    have h_sqrt_le : Real.sqrt T ≤ T := by
      rw [← Real.sqrt_sq hT0]
      exact Real.sqrt_le_sqrt (by nlinarith)
    linarith
  exact not_NvanLittleO_of_le_NR X δ (1 / 2) (by norm_num) hle hδ_nonneg hvan
